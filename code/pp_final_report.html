Research Question

What are the key factors that influence the estimated construction cost of building permits in Toronto from 2017 to 2024?

Load Dataset

building_data <- read.csv("../data/raw_building_data.csv", stringsAsFactors = FALSE)

Clean and Transform Data

# Convert EST_CONST_COST to numeric
building_data$EST_CONST_COST <- as.numeric(gsub(",", "", building_data$EST_CONST_COST))
# Filter out non-positive or missing EST_CONST_COST
building_data <- building_data[!is.na(building_data$EST_CONST_COST) & building_data$EST_CONST_COST > 0, ]
# Remove negative values in DWELLING_UNITS_CREATED and RESIDENTIAL
building_data <- building_data[building_data$DWELLING_UNITS_CREATED >= 0 & 
                               building_data$RESIDENTIAL >= 0, ]
# Select relevant columns
cleaned_data <- building_data[, c("EST_CONST_COST", "PERMIT_TYPE", "STRUCTURE_TYPE", 
                                  "WARD_GRID", "DWELLING_UNITS_CREATED", "RESIDENTIAL")]
# Drop any remaining NAs
cleaned_data <- na.omit(cleaned_data)
# Set baseline for PERMIT_TYPE
cleaned_data$PERMIT_TYPE <- relevel(as.factor(cleaned_data$PERMIT_TYPE), 
                                    ref = "Residential Building Permit")

df2 <- cleaned_data %>%
  mutate(
    log_cost      = log(EST_CONST_COST),
    PERMIT_TYPE   = fct_lump_n(as.factor(PERMIT_TYPE), n = 15, other_level = "Other"),
    STRUCTURE_TYPE= fct_lump_n(as.factor(STRUCTURE_TYPE), n = 15, other_level = "Other"),
    WARD_GRID     = fct_lump_n(as.factor(WARD_GRID), n = 30, other_level = "Other")
  )

Save Cleaned Data

write.csv(cleaned_data, "../data/cleaned_building_data.csv", row.names = FALSE)
